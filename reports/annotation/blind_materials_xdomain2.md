#!/usr/bin/env python3
"""
Train a multitask toxicity/FDA approval predictor for ClinTox and save test
positive-class probabilities to pred_results/clintox_test_pred.csv.

DeepChem/RDKit are not installed in this runtime, so this script implements a
small self-contained Morgan/ECFP-like hashed circular fingerprint from SMILES
(connectivity parsed for common organic/charged/bracket/ring/branch syntax) and
trains one-vs-task sklearn classifiers as a multitask surrogate.
"""

import re

TASKS = ["FDA_APPROVED", "CT_TOX"]
N_BITS = 2048
RADIUS = 2

def stable_hash_int(x: str) -> int:
    return int.from_bytes(hashlib.blake2b(x.encode("utf-8"), digest_size=8).digest(), "little", signed=False)

def tokenize_smiles(smiles: str):
    tokens = []
    i = 0
    two_char_atoms = {"Cl", "Br", "Si", "Na", "Li", "Al", "Ca", "Fe", "Mg", "Zn", "Cu", "Mn", "Hg", "Sn", "Ag", "Au", "Pt", "Pd"}
    aromatic = set("bcnops")
    while i < len(smiles):
        ch = smiles[i]
        if ch == "[":
            j = smiles.find("]", i + 1)
            if j == -1:
                tokens.append(ch)
                i += 1
            else:
                tokens.append(smiles[i:j+1])
                i = j + 1
        elif i + 1 < len(smiles) and smiles[i:i+2] in two_char_atoms:
            tokens.append(smiles[i:i+2])
            i += 2
        elif ch == "%" and i + 2 < len(smiles) and smiles[i+1:i+3].isdigit():
            tokens.append(smiles[i:i+3])
            i += 3
        else:
            tokens.append(ch)
            i += 1
    return tokens

def atom_symbol(tok: str) -> str:
    if tok.startswith("[") and tok.endswith("]"):
        inner = tok[1:-1]
        # isotope prefix, then element/aromatic symbol
        m = re.match(r"^\d*([A-Z][a-z]?|[a-z])", inner)
        if m:
            return m.group(1)
        return "*"
    return tok

def atom_charge(tok: str) -> int:
    if not tok.startswith("["):
        return 0
    inner = tok[1:-1]
    charge = 0
    for m in re.finditer(r"([+-]+)(\d*)", inner):
        signs, num = m.group(1), m.group(2)
        val = int(num) if num else len(signs)
        charge += val if signs[0] == "+" else -val
    return charge

def atom_hcount(tok: str) -> int:
    if not tok.startswith("["):
        return 0
    inner = tok[1:-1]
    m = re.search(r"H(\d*)", inner)
    if not m:
        return 0
    return int(m.group(1)) if m.group(1) else 1

def atom_aromatic(tok: str) -> int:
    sym = atom_symbol(tok)
    return int(sym.islower())

def parse_smiles_graph(smiles: str):
    toks = tokenize_smiles(smiles)
    atoms = []
    adj = []
    stack = []
    rings = {}
    current = None
    pending_bond = "-"
    bond_order_map = {"-": "1", "=": "2", "#": "3", ":": "a", "/": "1", "\\": "1"}
    bond_tokens = set(bond_order_map)

    def add_atom(tok):
        idx = len(atoms)
        atoms.append(tok)
        adj.append([])
        return idx

    i = 0
    while i < len(toks):
        tok = toks[i]
        if tok in bond_tokens:
            pending_bond = bond_order_map.get(tok, "1")
        elif tok == "(":
            stack.append(current)
        elif tok == ")":
            current = stack.pop() if stack else current
        elif tok == ".":
            current = None
            pending_bond = "-"
        elif tok.isdigit() or (tok.startswith("%") and tok[1:].isdigit()):
            if current is not None:
                r = tok[1:] if tok.startswith("%") else tok
                if r in rings:
                    other, b = rings.pop(r)
                    bond = pending_bond if pending_bond != "-" else b
                    adj[current].append((other, bond))
                    adj[other].append((current, bond))
                else:
                    rings[r] = (current, pending_bond if pending_bond != "-" else "1")
            pending_bond = "-"
        else:
            # Accept organic, aromatic, wildcard and bracket atoms; ignore any unknown punctuation.
            is_atom = tok.startswith("[") or tok == "*" or re.match(r"^[A-Z][a-z]?$", tok) or re.match(r"^[bcnops]$", tok)
            if is_atom:
                new = add_atom(tok)
                if current is not None:
                    # aromatic-aromatic unmarked bonds are aromatic
                    b = pending_bond
                    if b == "-" and atom_aromatic(atoms[current]) and atom_aromatic(tok):
                        b = "a"
                    elif b == "-":
                        b = "1"
                    adj[current].append((new, b))
                    adj[new].append((current, b))
                current = new
                pending_bond = "-"
        i += 1
    return atoms, adj

def ecfp_like_features(smiles_list, n_bits=N_BITS, radius=RADIUS):
    X = np.zeros((len(smiles_list), n_bits), dtype=np.float32)
    parse_fail = 0
    atom_counts = []
    for row, smi in enumerate(smiles_list):
        atoms, adj = parse_smiles_graph(str(smi))
        atom_counts.append(len(atoms))
        if not atoms:
            parse_fail += 1
            continue
        invariants = []
        for i, tok in enumerate(atoms):
            sym = atom_symbol(tok)
            inv = f"{sym}|ar{atom_aromatic(tok)}|chg{atom_charge(tok)}|H{atom_hcount(tok)}|deg{len(adj[i])}"
            invariants.append(str(stable_hash_int(inv)))
        identifiers = invariants[:]
        # radius 0
        for ident in identifiers:
            X[row, stable_hash_int("r0|" + ident) % n_bits] += 1.0
        for r in range(1, radius + 1):
            new_identifiers = []
            for i, ident in enumerate(identifiers):
                neigh = sorted((bond, identifiers[j]) for j, bond in adj[i])
                env = f"r{r}|{ident}|" + "|".join(f"{b}:{nid}" for b, nid in neigh)
                new_id = str(stable_hash_int(env))
                new_identifiers.append(new_id)
                X[row, stable_hash_int(env) % n_bits] += 1.0
            identifiers = new_identifiers

        # Add salt/size/global descriptors into tail via hashing as counts.
        chars = str(smi)
        for key, val in {
            "num_atoms": len(atoms),
            "num_bonds": sum(len(a) for a in adj) // 2,
            "num_components": chars.count(".") + 1,
            "len_bin": min(len(chars) // 10, 30),
            "charge_plus": chars.count("+"),
            "charge_minus": chars.count("-"),
        }.items():
            X[row, stable_hash_int(f"global|{key}|{val}") % n_bits] += 1.0
    # log-count transform (binary/count hybrid) to reduce repeated-substructure dominance
    X = np.log1p(X)
    print(f"Featurized {len(smiles_list)} molecules; parse_fail={parse_fail}; atom_count min/median/max={np.min(atom_counts)}/{np.median(atom_counts):.1f}/{np.max(atom_counts)}")
    return X

def safe_auc(y, p):
    try:
        if len(np.unique(y)) < 2:
            return np.nan
        return roc_auc_score(y, p)
    except Exception:
        return np.nan

def train_task_ensemble(X, y, seed=123):
    # Use a weighted blend of diverse sklearn classifiers.  Class imbalance is severe.
    models = []
    # Logistic on sparse-ish hashed circular features, class-balanced
    models.append(("logreg", make_pipeline(
        StandardScaler(with_mean=False),
        LogisticRegression(max_iter=3000, class_weight="balanced", solver="liblinear", C=0.5, random_state=seed)
    )))
    models.append(("extratrees", ExtraTreesClassifier(
        n_estimators=700, max_features="sqrt", min_samples_leaf=1,
        class_weight="balanced_subsample", random_state=seed, n_jobs=-1
    )))
    models.append(("rf", RandomForestClassifier(
        n_estimators=500, max_features="sqrt", min_samples_leaf=1,
        class_weight="balanced_subsample", random_state=seed+1, n_jobs=-1
    )))
    fitted = []
    for name, model in models:
        model.fit(X, y)
        fitted.append((name, model))
    return fitted

def predict_ensemble(fitted, X):
    probs = []
    names = []
    for name, model in fitted:
        p = model.predict_proba(X)
        if p.shape[1] == 1:
            # Degenerate fallback, should not occur in training labels here.
            pp = np.full(X.shape[0], float(model.classes_[0]))
        else:
            pos_idx = list(model.classes_).index(1)
            pp = p[:, pos_idx]
        probs.append(pp)
        names.append(name)
    return np.mean(probs, axis=0), dict(zip(names, probs))

def main():
    out_dir = Path("pred_results")
    out_dir.mkdir(exist_ok=True)
    train = pd.read_csv("clintox/clintox_train.csv")

    X = ecfp_like_features(train["smiles"].tolist())
    X_test = ecfp_like_features(test["smiles"].tolist())
    y_all = train[TASKS].astype(int).values

    # Internal validation by a stratified split on the joint multitask labels.
    joint = train[TASKS].astype(str).agg("_".join, axis=1)
    # Rare joint class may be small but is >=14/82 for positives; stratified split works.
    splitter = StratifiedShuffleSplit(n_splits=1, test_size=0.2, random_state=77)
    tr_idx, va_idx = next(splitter.split(X, joint))
    print("Validation split sizes:", len(tr_idx), len(va_idx))

    val_summary = {}
    for j, task in enumerate(TASKS):
        fitted_val = train_task_ensemble(X[tr_idx], y_all[tr_idx, j], seed=200+j)
        pv, comps = predict_ensemble(fitted_val, X[va_idx])
        yv = y_all[va_idx, j]
        val_summary[task] = {
            "roc_auc": float(safe_auc(yv, pv)) if not math.isnan(safe_auc(yv, pv)) else None,
            "avg_precision": float(average_precision_score(yv, pv)) if len(np.unique(yv)) > 1 else None,
            "log_loss": float(log_loss(yv, np.clip(pv, 1e-6, 1-1e-6), labels=[0,1])),
            "positive_rate_val": float(yv.mean()),
            "pred_mean": float(pv.mean()),
            "pred_min": float(pv.min()),
            "pred_max": float(pv.max()),
        }
        print("VAL", task, val_summary[task])

    # Fit final models on all training data and predict test positive-label probabilities.
    pred = pd.DataFrame({"smiles": test["smiles"].astype(str)})
    all_component_stats = {}
    for j, task in enumerate(TASKS):
        fitted = train_task_ensemble(X, y_all[:, j], seed=500+j)
        ptest, comp = predict_ensemble(fitted, X_test)
        # Mild clipping for valid probabilities away from exact endpoints.
        ptest = np.clip(ptest, 1e-6, 1-1e-6)
        pred[task] = ptest
        all_component_stats[task] = {k: {"mean": float(v.mean()), "min": float(v.min()), "max": float(v.max())} for k, v in comp.items()}
        print("TEST", task, "mean/min/max", float(ptest.mean()), float(ptest.min()), float(ptest.max()))

    out_path = out_dir / "clintox_test_pred.csv"
    pred.to_csv(out_path, index=False)
    print("Saved:", out_path, "shape:", pred.shape)
    print(pred.head(10).to_string(index=False))
    print("Validation summary JSON:", pd.Series(val_summary).to_json())
    print("Component stats:", all_component_stats)

if __name__ == "__main__":
    main()

def parse_abc_angles(structure_text):
    abc = [np.nan, np.nan, np.nan]
    ang = [np.nan, np.nan, np.nan]
    nsites = np.nan
    m = re.search(r"abc\s*:\s*([0-9.Ee+-]+)\s+([0-9.Ee+-]+)\s+([0-9.Ee+-]+)", str(structure_text))
    if m:
        abc = [float(m.group(i)) for i in range(1, 4)]
    m = re.search(r"angles\s*:\s*([0-9.Ee+-]+)\s+([0-9.Ee+-]+)\s+([0-9.Ee+-]+)", str(structure_text))
    if m:
        ang = [float(m.group(i)) for i in range(1, 4)]
    m = re.search(r"Sites\s*\((\d+)\)", str(structure_text))
    if m:
        nsites = int(m.group(1))
    return abc + ang + [nsites]

def formula_counts(formula):
    # Simple parser adequate for reduced formulas like CoPt3, Mg3Cd; no parentheses in preview.
    toks = re.findall(r"([A-Z][a-z]?)([0-9]*\.?[0-9]*)", str(formula))
    d = {}
    for el, num in toks:
        val = float(num) if num else 1.0
        d[el] = d.get(el, 0.0) + val
    return d

def make_features(df):
    out = pd.DataFrame(index=df.index)
    out["space_group"] = df["space_group"].astype(float)
    out["elastic_anisotropy"] = df["elastic_anisotropy"].astype(float)
    out["G_VRH"] = df["G_VRH"].astype(float)
    out["poisson_ratio"] = df["poisson_ratio"].astype(float)
    denom = 3 * (1 - 2 * out["poisson_ratio"])
    out["K_from_G_nu"] = 2 * out["G_VRH"] * (1 + out["poisson_ratio"]) / denom.replace(0, np.nan)
    out["log_G"] = np.log1p(np.maximum(out["G_VRH"], 0))
    out["sg_crystal_system"] = pd.cut(
        out["space_group"],
        bins=[0,2,15,74,142,167,194,230],
        labels=False,
        include_lowest=True
    ).astype(float)
    
    parsed = np.array([parse_abc_angles(s) for s in df["structure"]], dtype=float)
    for i, name in enumerate(["a","b","c","alpha","beta","gamma","nsites"]):
        out[name] = parsed[:, i]
    a,b,c = out["a"], out["b"], out["c"]
    al,be,ga = np.deg2rad(out["alpha"]), np.deg2rad(out["beta"]), np.deg2rad(out["gamma"])

    out["cell_volume"] = volume
    out["volume_per_site"] = volume / out["nsites"].replace(0, np.nan)
    out["abc_mean"] = (a+b+c)/3
    out["abc_std"] = np.vstack([a,b,c]).std(axis=0)
    out["abc_max_min_ratio"] = np.maximum.reduce([a,b,c]) / np.minimum.reduce([a,b,c])
    
    comp_rows = []
    for f in df["formula"]:
        counts = formula_counts(f)
        total = sum(counts.values()) if counts else np.nan

        fracs = np.array([counts[e]/total for e in els], dtype=float) if total and not np.isnan(total) else np.array([])
