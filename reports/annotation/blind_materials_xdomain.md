#!/usr/bin/env python3
import os
import json
import warnings
import numpy as np
import pandas as pd

from sklearn.ensemble import RandomForestClassifier
from sklearn.impute import SimpleImputer
from sklearn.model_selection import StratifiedKFold, RandomizedSearchCV, cross_val_score
from sklearn.pipeline import Pipeline
from sklearn.metrics import balanced_accuracy_score, make_scorer
from scipy.stats import randint

warnings.filterwarnings("ignore", category=UserWarning)
warnings.filterwarnings("ignore", category=RuntimeWarning)

RANDOM_STATE = 20251010
os.makedirs("pred_results", exist_ok=True)

desc_path = "dili_MD/mol_descriptors_training.csv"
meta_path = "dili_MD/standardized_compounds_excl_ambiguous_cluster.csv"
test_path = "dili_MD/test.csv"

desc = pd.read_csv(desc_path)
meta = pd.read_csv(meta_path)
test = pd.read_csv(test_path)

print("Loaded:")
print("  descriptors:", desc.shape)
print("  metadata:", meta.shape)
print("  test:", test.shape)
print("vDILIConcern counts:")
print(meta["vDILIConcern"].value_counts(dropna=False).to_string())

# Align descriptor matrices. The first column is an index-like column named Unnamed: 0.
id_col = desc.columns[0]
feature_cols = [c for c in desc.columns if c != id_col]

    raise RuntimeError("Test set is missing training descriptor columns: " + str(missing_in_test[:10]))

    print("Dropping all-NaN columns:", len(all_nan_cols), all_nan_cols[:10])
    X_all = X_all.drop(columns=all_nan_cols)
    X_test = X_test.drop(columns=all_nan_cols)

# Binary split definitions inferred from DILI benchmark naming:

    "MCNC": {
        "positive": {"vMost-DILI-Concern"},
        "negative": {"vNo-DILI-Concern"},
        "include": {"vMost-DILI-Concern", "vNo-DILI-Concern"},
    },
    "MCLCNC": {

        "negative": None,
        "include": None,

    "rf__min_samples_split": randint(2, 16),
    "rf__min_samples_leaf": randint(1, 8),
    "rf__bootstrap": [True],
    "rf__criterion": ["gini", "entropy"],
}

summary = {}

    labels_raw = meta["vDILIConcern"].astype(str)
    if spec["include"] is None:

        mask = labels_raw.isin(spec["include"]).to_numpy()

        verbose=0,
        error_score="raise",

    best_model = search.best_estimator_
    cv_scores = cross_val_score(best_model, X, y, cv=cv, scoring=make_scorer(balanced_accuracy_score), n_jobs=-1)

    print("Best params:", search.best_params_)
    print("Refit model CV balanced accuracy mean±sd:", round(float(cv_scores.mean()), 4), "±", round(float(cv_scores.std()), 4))

    pred = best_model.predict(X_test)
    pred_labels = np.where(pred == 1, "DILI", "NoDILI")

    out = test.copy()
    out["label"] = pred_labels

    label_counts = pd.Series(pred_labels).value_counts().to_dict()
    print("Saved:", out_path, "shape:", out.shape, "prediction counts:", label_counts)

        "n_train": int(X.shape[0]),
        "n_features": int(X.shape[1]),
        "binary_counts": {"NoDILI": int((y == 0).sum()), "DILI": int((y == 1).sum())},
        "best_score_balanced_accuracy": float(search.best_score_),
        "cv_mean_balanced_accuracy": float(cv_scores.mean()),
        "cv_sd_balanced_accuracy": float(cv_scores.std()),
        "best_params": search.best_params_,
        "prediction_counts": {str(k): int(v) for k, v in label_counts.items()},
        "output": out_path,
    }

with open("pred_results/RF_training_summary.json", "w") as f:
    json.dump(summary, f, indent=2)

#!/usr/bin/env python3
from pathlib import Path
import pandas as pd
from collections import Counter, defaultdict

base = Path("nvc")
rag = pd.read_csv(base/"Ragni2016.csv")

def mood_letter(q):
    return {"All":"A", "Some":"I", "Some not":"O", "No":"E"}[q]

def parse_statement(s):
    parts = s.split(";")
    if len(parts) != 3:
        raise ValueError(f"bad statement {s!r}")
    return parts[0], parts[1], parts[2]

def get_end_terms_from_choices(choices):
    opts = choices.split("|")
    terms = []
    for o in opts:
        if o == "NVC":
            continue
        q, s, p = parse_statement(o)
        terms.extend([s,p])
        # first option is All;a;c by dataset construction
        return s, p
    raise ValueError("no non-NVC choice")

def canonical_task(task, choices):
    p1s, p2s = task.split("/")
    q1, s1, p1 = parse_statement(p1s)
    q2, s2, p2 = parse_statement(p2s)
    a, c = get_end_terms_from_choices(choices)
    # middle is the term not in {a,c}
    terms = [s1,p1,s2,p2]
    mids = [t for t in terms if t not in {a,c}]
    if not mids:
        raise ValueError(f"no middle for {task} choices {choices}")
    m = Counter(mids).most_common(1)[0][0]
    # Figure convention used by common syllogistic model files:
    # 1: a-m / m-c; 2: m-a / c-m; 3: a-m / c-m; 4: m-a / m-c
    pos1 = ("S" if s1 == m else "P")  # where is middle in first premise?
    pos2 = ("S" if s2 == m else "P")  # where is middle in second premise?
    figmap = {("P","S"):"1", ("S","P"):"2", ("P","P"):"3", ("S","S"):"4"}
    fig = figmap[(pos1,pos2)]
    return mood_letter(q1) + mood_letter(q2) + fig

def canonical_response(resp, choices):
    if resp == "NVC":
        return "NVC"
    q, s, p = parse_statement(resp)
    a, c = get_end_terms_from_choices(choices)
    if s == a and p == c:
        direction = "ac"
    elif s == c and p == a:
        direction = "ca"
    else:
        raise ValueError(f"response terms do not match choices: {resp}, {choices}")
    return mood_letter(q) + direction

rag["syllog"] = [canonical_task(t, ch) for t, ch in zip(rag["task"], rag["choices"])]
rag["canon_response"] = [canonical_response(r, ch) for r, ch in zip(rag["response"], rag["choices"])]
